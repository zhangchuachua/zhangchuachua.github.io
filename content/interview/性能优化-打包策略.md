## webpack

```js
const path = require('path');
const HtmlWebpackPlugin = require('html-webpack-plugin');
const CssMinimizerPlugin = require('css-minimizer-webpack-plugin');
const MiniCssExtractPlugin = require('mini-css-extract-plugin');

/** @type {import('webpack').Configuration} */
module.exports = {
  mode: process.env.NODE_ENV,
  entry: './src/index.js',
  output: {
    filename: '[name].js',
    // chunkFilename 只适用于异步加载的 chunk 文件，例如 import() 引入的
    chunkFilename: '[name].[contenthash:8].chunk.js',
    path: path.resolve(__dirname, 'dist'),
    clean: true,
  },
  devServer: {
    static: './dist',
    port: 3001,
    open: false,
  },
  module: {
    rules: [
      {
        test: /\.(js|jsx)$/,
        exclude: /node_modules/,
        use: [
          {
            loader: 'babel-loader',
            options: {
              presets: [
                '@babel/preset-env',
                [
                  '@babel/preset-react',
                  {
                    runtime: 'automatic',
                  }
                ]]
            }
          }
        ]
      },
      {
        test: /\.css$/,
        use: [
          // 'style-loader', // 把 css 打包到 js 中，然后通过 js 创建 style 标签，将 css 写入, 主要用于开发
          MiniCssExtractPlugin.loader,
          'css-loader',// 先执行 css-loader 再执行上面的 loader
        ],
      },
      // {
      //   test: /\.svg$/,
      //   // *resourceQuery: 'react', // 这个点作用就是当遇到 svg?react 的时候将会使用这个 rules
      //   // *layer: 'css', // 当在 css 中引入 svg 时将会使用这个 rules
      //   // *issuerLayer: 'style', // 当 `import './icon.svg' with { layer: 'style' }` 这样引入时，将会使用这个规则
      //   use: [
      //     '@svgr/webpack',
      //   ]
      // }
    ]
  },
  plugins: [
    new HtmlWebpackPlugin({
      template: './public/index.html',
    }),
    new MiniCssExtractPlugin({ filename: 'index.css', /* 默认为 main.css */ }),// 将 css 文件提取
  ],
  optimization: {
    // 不太重要 deterministic 表示生成的 id 以一种更确定的方式生成，而不是随机生成。这样的话当内容没有变化时，id也保持不变，提高缓存的利用率
    moduleIds: 'deterministic',
    chunkIds: 'deterministic',
    minimize: true,
    minimizer: [
      '...',
      new CssMinimizerPlugin(),
    ],
    splitChunks: {
      chunks: 'all',
      maxInitialRequests: Infinity,// 入口能接受的最大请求
      // *下面这两个是全局规则，并且是按照未压缩之前的体积算的, 也就是说如果压缩之前的体积大于 250kb 那么就会进行切割
      minSize: 20000,// 20KB 避免过小请求
      maxSize: 250000, // 250KB 避免单包过大，如果多个包打包结果大于 250kb 那么将会进行切割，如果有一个组件本身就大于 250kb 那么不会进行切割。
      cacheGroups: {
        // 1. react vue 等核心框架，变更频率很低，单独分包可以长期缓存
        react: {
          test: /[\\/]node_modules[\\/](react|react-dom)/,
          name: 'react-core',
          priority: 40,
          // 可以覆盖全局规则
          maxSize: 4000000,
          enforce: true
        },
        // 2. ui 库单独分包, 独立于业务逻辑，避免业务代码变更影响 UI 缓存，所以单独分包
        uiLibrary: {
          test: /[\\/]node_modules[\\/](antd|@material-ui|bootstrap)[\\/]/,
          name: 'ui-lib',
          priority: 30
        },
        // 3. 工具库 体积中等，但是复用率高，可能被高频使用所以单独打包
        utilityLib: {
          test: /[\\/]node_modules[\\/](lodash|moment|dayjs|axios|ramda)[\\/]/,
          name: 'utility-lib',
          priority: 20
        },
        // 4. 公共模块 被多次引用的模块，为了避免加载相同代码，单独打包
        commons: {
          name: 'commons',
          minChunks: 2, // 被2个以上入口引用
          priority: 10,
          // 该模块被多次引用时将会复用这个包 而不是重新生成, 如果设置为 false 那么将会重复打包
          reuseExistingChunk: true,
          // 可以覆盖全局的配置，将会覆盖 minSize: 20000, 所以即使有一个公共包体积小于 20kb 也会被打包。
          minSize: 0,
          chunks: 'all'
        },
        // 默认第三方库 兜底
        vendor: {
          test: /[\\/]node_modules[\\/]/,
          name: 'vendors',
          priority: 0
        },
      },
    },
    // 将Webpack的运行时代码单独拆分，避免因模块增减导致入口文件变化（影响缓存）。
    runtimeChunk: 'single'
  },
  resolve: {
    extensions: ['.js', '.jsx'],
  }
}
```
