## item 高度确定的情况

```jsx
import { useMemo, useState } from "react"
import { useCallback } from "react"

// 虚拟列表有几个重要的点：
// 1. container 的高度，用于计算显示的 item 个数
// 2. item 的高度 + item 的数量 计算出总高度
// 3. scrollTop 计算 startIndex
// 4. startIndex 计算 endIndex， 再由 startIndex 和 endIndex 计算出需要渲染的 item
export default function FixedSizeList({ itemHeight = 50, itemSumCount = 100, containerHeight = 200, buffer = 5 }) {
  const [allHeight, range] = useMemo(() => {
    return [itemHeight * itemSumCount, Math.ceil(containerHeight / itemHeight)];
  }, [itemHeight, itemSumCount, containerHeight]);

  const [startIndex, setStartIndex] = useState(0);
  const [endIdx, setEndIdx] = useState(startIndex + range + buffer);
  const handleScroll = useCallback((e) => {
    const scrollTop = e.currentTarget.scrollTop
    const offset = Math.floor(scrollTop / itemHeight);
    setStartIndex(
      Math.max(
        0,
        offset - buffer
      )
    )
    setEndIdx(
      Math.min(
        itemSumCount,
        offset + range + buffer
      )
    )
  }, [])


  return <div className="w-1080px h-200px mx-a overflow-auto" onScroll={handleScroll}>
    <div className="w-full bg-red relative" style={{ height: allHeight + 'px' }}>
      <Row start={startIndex} end={endIdx}></Row>
    </div>
  </div>
}

const Row = ({ start, end }) => {
  const rows = Array.from({ length: end - start }, (_, i) => i + start)
  return rows.map((i) => <div key={i} className="w-full h-45px mb-5px bg-blue absolute inset-0"
    style={{ transform: `translateY(${i * 50}px)` }}>{i}</div>)
}
```

## item 高度不确定的情况

```jsx
import React, { useCallback, useMemo, useState, useEffect } from "react"
import ResizeObserver from 'rc-resize-observer';
import useHeights from "../hooks/useHeights.js";

// 参考于 rc-virtual-list
// 重点是要计算： start, end 和 offset
// 在不知道 item 高度的情况下，第一次渲染时，就是用 item 的默认高度去进行计算；
// 当 item 渲染完成，收集每个 item 的 dom 元素，然后通过 dom 元素获取真实的高度；
// 再通过真实的高度去重新计算 start, end, offset 和 真实的高度
export default function VariableSizeList(
  {
    itemHeight = 30,// 默认高度
    buffer = 5,
    height = 200, // 容器高度
    getKey,
    data,
    children,
    onScroll,
    scrollToIndex,
  }) {

  const [setElementsRef, collectHeight, heights, heightUpdatedMark] = useHeights(getKey);
  const [scrollTop, setScrollTop] = useState(0);

  // !重点：计算
  const { start, end, scrollHeight, offset } = useMemo(() => {
    const len = data.length;
    let itemTop = 0;
    let bufferItemTop = [];
    let startIndex;
    let startOffset;
    let endIndex;

    for (let i = 0; i < len; i++) {
      const item = data[i];
      const key = item.id;
      const cacheHeight = heights.get(key);
      const currentItemHeight = cacheHeight === undefined ? itemHeight : cacheHeight;

      if (bufferItemTop.length <= buffer) {
        bufferItemTop.push(currentItemHeight);
      } else {
        bufferItemTop.shift();
        bufferItemTop.push(currentItemHeight);
      }

      const currentItemBottom = itemTop + currentItemHeight;

      // !实现 buffer 的重点！！
      if (currentItemBottom >= scrollTop && startIndex === undefined) {
        startIndex = i - buffer;
        if (startIndex <= 0) startOffset = 0;
        else startOffset = currentItemBottom - bufferItemTop.reduce((a, b) => a + b, 0);
      }
      if (currentItemBottom > scrollTop + height && endIndex === undefined) {
        endIndex = i;
      }
      itemTop = currentItemBottom;

    }

    // When scrollTop at the end but data cut to small count will reach this
    if (startIndex === undefined) {
      startIndex = 0;
      startOffset = 0;
      endIndex = Math.ceil(height / itemHeight);
    }
    if (endIndex === undefined) {
      endIndex = data.length - 1;
    }

    startIndex = Math.max(startIndex, 0);
    endIndex = Math.min(endIndex + 1 + buffer, data.length);

    return {
      start: startIndex,
      end: endIndex,
      scrollHeight: itemTop,
      offset: startOffset,
    }
  }, [height, data, itemHeight, scrollTop, heightUpdatedMark]);

  // 添加滚动到指定索引的功能
  useEffect(() => {
    if (typeof scrollToIndex === 'number') {
      let offset = 0;
      for (let i = 0; i < scrollToIndex; i++) {
        const item = data[i];
        const key = getKey(item);
        const cacheHeight = heights.get(key);
        offset += cacheHeight === undefined ? itemHeight : cacheHeight;
      }
      setScrollTop(offset);
    }
  }, [scrollToIndex, data, heights, itemHeight]);

  const handleScroll = useCallback((e) => {
    const scrollTop = e.currentTarget.scrollTop;
    setScrollTop(scrollTop);
    onScroll?.(e);
  }, [onScroll]);

  const onResize = useCallback(({ offsetHeight }) => {
    if (offsetHeight) {
      collectHeight();
    }
  }, [collectHeight]);

  return <div className="w-1080px mx-a overflow-auto bg-red" onScroll={handleScroll} style={{ height }}>
    <div className={"relative overflow-hidden"} style={{ height: scrollHeight + 'px' }}>
      <ResizeObserver onResize={onResize}>
        <div className={'absolute left-0 right-0 flex flex-col'} style={{ transform: `translateY(${offset}px)` }}>
          <Items
            start={start} end={end} setElementsRef={setElementsRef} renderFunc={children}
            getKey={getKey}>{data}</Items>
        </div>
      </ResizeObserver>
    </div>
  </div>
}

const Items = ({ start, end, children, setElementsRef, renderFunc, getKey }) => {
  return children.slice(start, end).map((item) => {
    return <div key={item.id} ref={(ele) => {
      setElementsRef(item, ele)
    }} className={"shrink-0"}>
      {
        // 需要啥 prop 可以自定义
        renderFunc(item)
      }
    </div>
  })
}
```

```js
import React from 'react';
import { useEffect, useRef } from 'react';

function parseNumber(value) {
  const num = parseFloat(value);
  return isNaN(num) ? 0 : num;
}

export default function useHeights(
  getKey,
) {
  const [updatedMark, setUpdatedMark] = React.useState(0);
  const elementsRef = useRef(new Map());
  const heightsRef = useRef(new Map());

  const promiseIdRef = useRef(0);

  function cancelRaf() {
    promiseIdRef.current += 1;
  }

  function collectHeight(sync = false) {
    cancelRaf();

    const doCollect = () => {
      let changed = false;

      elementsRef.current.forEach((element, key) => {
        if (element && element.offsetParent) {
          const htmlElement = element;
          const { offsetHeight } = htmlElement;
          const { marginTop, marginBottom } = getComputedStyle(htmlElement);

          const marginTopNum = parseNumber(marginTop);
          const marginBottomNum = parseNumber(marginBottom);
          const totalHeight = offsetHeight + marginTopNum + marginBottomNum;

          if (heightsRef.current.get(key) !== totalHeight) {
            heightsRef.current.set(key, totalHeight);
            changed = true;
          }
        }
      });

      // Always trigger update mark to tell parent that should re-calculate heights when resized
      if (changed) {
        setUpdatedMark((c) => c + 1);
      }
    };

    if (sync) {
      doCollect();
    } else {
      promiseIdRef.current += 1;
      const id = promiseIdRef.current;
      // 异步收集，这样可以等待 ref 都设置完成之后再收集，优化性能
      Promise.resolve().then(() => {
        if (id === promiseIdRef.current) {
          doCollect();
        }
      });
    }
  }

  function setElementsRef(item, instance) {
    const key = getKey(item);
    // 需要对应 node 删除或增加的情况时可以使用这个；
    const origin = elementsRef.current.get(key);

    if (instance) {
      elementsRef.current.set(key, instance);
      collectHeight();
    } else {
      elementsRef.current.delete(key);
    }
  }

  useEffect(() => {
    // 清理时去取消还没有执行的 doCollect
    return cancelRaf;
  }, []);

  return [setElementsRef, collectHeight, heightsRef.current, updatedMark];
}
```
